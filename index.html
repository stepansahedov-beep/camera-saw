Модуль интеграции: camerA_telegram.py Этот компонент отвечает исключительно за связь с API мессенджера. Он принимает сформированные сервером данные и отправляет их конечному администратору. python import requests TOKEN = "ВАШ_ТОКЕН_БОТА" CHAT_ID = "ВАШ_ЧАТ_ID" def send_data(text_message, photo_bytes=None): """Отправляет текстовый лог и фото (если есть) в Telegram чат""" try: if photo_bytes: # Метод отправки фото с подписью url = f"https://telegram.org{TOKEN}/sendPhoto" payload = {"chat_id": CHAT_ID, "caption": text_message} files = {"photo": ("image.jpg", photo_bytes, "image/jpeg")} requests.post(url, data=payload, files=files) else: # Метод отправки обычного текстового сообщения (если фото нет) url = f"https://telegram.org{TOKEN}/sendMessage" payload = {"chat_id": CHAT_ID, "text": text_message} requests.post(url, data=payload) except Exception as e: print(f"Ошибка отправки данных в Telegram: {e}")
